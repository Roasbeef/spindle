cmake_minimum_required(VERSION 3.24)
project(spindle_helper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(SPINDLE_METAL "Build the Metal backend" OFF)
set(GGML_METAL ${SPINDLE_METAL} CACHE BOOL "" FORCE)
FetchContent_Declare(llama_cpp
  URL https://github.com/ggml-org/llama.cpp/archive/427291b5b34cd914a31b3fd3b61a68f6184f4b9f.tar.gz
  URL_HASH SHA256=056ae92c363e5c761e6c03102f4aa6c1551e134f9e322eea1a4cf3f6cf87f585
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(llama_cpp)
find_package(Threads REQUIRED)
add_executable(spindle-helper main.cpp)
target_link_libraries(spindle-helper PRIVATE llama Threads::Threads)
target_compile_options(spindle-helper PRIVATE -Wall -Wextra -Werror)
install(TARGETS spindle-helper RUNTIME DESTINATION bin)
